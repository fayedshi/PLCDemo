1. 为什么直接 return（HTTP）在实时上位机中行不通？
巨大的性能开销（握手太重）：HTTP 是无状态的。每次前端向 Python 要数据，浏览器和后端都要经历一次“建立 TCP 连接 ->
 发送一大堆 HTTP 协议头（Headers）-> Python 处理 -> 返回数据 -> 断开连接”。在 100ms 的频率下，
这意味着一秒钟要进行 10 次完整的建立和断开连接。工控机的 CPU 会瞬间被这种无意义的“网络握手”占满。

2. 为什么 WebSocket 是实时波形的完美解法？
WebSocket 是一种全双工、长连接的协议。
它的工作流程是：只需一次握手：网页刚打开时，Vue 向 Python 发起一次 WebSocket 连接。一旦连接成功，这条通道就永远保持打开状态，直到你关闭网页。
后端主动推（Push）：Python 的采集线程一旦从 PLC 读到了新数据，不需要等待 Vue 来问，直接顺着这条管道把数据“滋”到前端（通过 await websocket.send_json()）。
极低的开销：因为通道一直开着，每次传数据只有纯粹的寄存器数值，没有任何多余的 HTTP 协议头。

influxdb可以存一般的数据吗InfluxDB 可以存一般的数据，但它并不是为此设计的，强行使用会导致严重的性能和存储问题。
InfluxDB 是一款专门的时序数据库（Time Series Database）。
它要求存储的数据必须包含一个时间戳（Timestamp）。为了让您直观了解，我们可以把“一般数据”分为两类来看 InfluxDB 的支持情况：
1. 带有时间属性的“一般数据”（完全可以存）如果您的业务数据是随着时间推移不断产生的，即使它看起来像普通的业务记录，InfluxDB 也能高效处理。
可以存的例子：用户的登录日志、电商系统的每笔订单生成记录、网站的页面访问（PV）流水、应用报错日志。优势：写入速度极快，自动按时间分区，支持自动过期清理（TTL），
非常适合做报表统计和趋势分析。2. 传统关系型的“一般数据”（绝对不建议存）如果您的数据是需要频繁修改、删除，或者属于实体状态的数据，
InfluxDB 会非常难用。不适合存的例子：用户账户资料（姓名、密码、余额）、商品库存信息、员工花名册、权限配置表。为什么不行？没有传统的 UPDATE 和 DELETE：
InfluxDB 的设计是“只追加写入”。虽然可以覆盖或删除，但代价极高，会严重拖慢数据库性能。不支持多表关联（Join）：它无法像 MySQL 那样通过外键把用户表和订单表关联起来。高基数（High Cardinality）灾难：如果把经常变动的值（如身份证号、唯一的订单号）作为 InfluxDB 的标签（Tag），
会导致内存暴涨甚至死机

import time
from pyModbusTCP.client import ModbusClient
from influxdb_client import InfluxDBClient, Point, WritePrecision
from influxdb_client.client.write_api import SYNCHRONOUS

# ==================== 1. 配置参数 ====================
# Modbus TCP 配置
PLC_IP = '127.0.0.1'
PLC_PORT = 502

# InfluxDB 配置（请根据您的实际后台填写）
INFLUX_URL = "http://localhost:8086"
INFLUX_TOKEN = "这里替换成你从InfluxDB后台生成的Token"
INFLUX_ORG = "你的组织名称"
INFLUX_BUCKET = "你的存储桶名称"

# ==================== 2. 初始化连接 ====================
# 初始化 Modbus
modbus_client = ModbusClient(host=PLC_IP, port=PLC_PORT, auto_open=True)

# 初始化 InfluxDB 客户端
influx_client = InfluxDBClient(url=INFLUX_URL, token=INFLUX_TOKEN, org=INFLUX_ORG)
# 激活写入 API (使用同步写入模式，适合入门调试)
write_api = influx_client.write_api(write_options=SYNCHRONOUS)

print("正在启动上位机采集与存储服务...")

# ==================== 3. 核心循环采集 ====================
try:
    while True:
        # 读取保持寄存器（从地址 0 开始，连续读 3 个）
        # 假设：regs[0]是温度，regs[1]是压力，regs[2]是产量
        regs = modbus_client.read_holding_registers(0, 3)
        
        if regs:
            temp_val = regs[0]
            press_val = regs[1]
            count_val = regs[2]
            print(f"成功读取PLC -> 温度:{temp_val}, 压力:{press_val}, 产量:{count_val}")
            
            # 创建一个 InfluxDB 数据点 (Point)
            # measurement 类似于表名（如设备名 device_01）
            # tag 类似于索引（用于分类筛选，如车间号）
            # field 是具体的物理量（数值数据）
            point = Point("device_status") \
                .tag("workshop", "line_A") \
                .field("temperature", float(temp_val)) \
                .field("pressure", float(press_val)) \
                .field("production_count", int(count_val)) \
                .time(time.time_ns(), WritePrecision.NS) # 自动打上当前纳米级时间戳
            
            try:
                # 写入数据库
                write_api.write(bucket=INFLUX_BUCKET, org=INFLUX_ORG, record=point)
                print(" -> 数据已成功存入 InfluxDB")
            except Exception as db_err:
                print(f" -> 数据库写入失败: {db_err}")
                
        else:
            print("PLC 读取失败，请检查网络...")
            
        # 采集频率：每隔 1 秒采集并存储一次
        time.sleep(1)

except KeyboardInterrupt:
    print("\n程序已手动停止。")
    # 关闭连接，释放资源
    modbus_client.close()
    write_api.close()
    influx_client.close()

const initWebSocket = () => {
  // 如果在电脑本机测试，保持 localhost；如果要手机访问，请改为工控机的局域网 IP
  socket = new WebSocket('ws://localhost:8000/ws/live');

  // 发生错误事件
  socket.onerror = (error) => {
    console.error('【前端提示】WebSocket 发生错误:', error);
  };
};

onBeforeUnmount(() => {

});

import { ref, onMounted, onBeforeUnmount, reactive } from 'vue';
import * as echarts from 'echarts';

// 1. 响应式变量与数据缓存
const chartRef = ref(null);
let myChart = null;
const currentData = reactive({ temperature: 0.0, pressure: 0.0, count: 0 });

// 内存中维护的实时数组（用于图表滚动）
const timeData = [];
const tempData = [];
const pressData = [];
const MAX_POINTS = 100; // 图表最多显示最近的 100 个数据点，防止内存溢出

let ws = null;

// 2. 初始化 ECharts 配置
const initChart = () => {
  if (!chartRef.value) return;
  myChart = echarts.init(chartRef.value, 'dark'); // 使用内置暗黑科技感主题

  const option = {
    title: { text: '设备实时参数监控 (100ms)', left: 'center', textStyle: { color: '#00ffcc' } },
    tooltip: { trigger: 'axis', axisPointer: { type: 'cross' } },
    legend: { data: ['温度 (℃)', '压力 (Mpa)'], top: '10%' },
    grid: { left: '5%', right: '5%', bottom: '10%', containLabel: true },
    xAxis: {
      type: 'category',
      boundaryGap: false,
      data: timeData,
      axisLine: { lineStyle: { color: '#444' } }

      { type: 'value', name: '温度', min: 0, max: 100, axisLabel: { formatter: '{value} ℃' } },
      { type: 'value', name: '压力', min: 0, max: 10, position: 'right', axisLabel: { formatter: '{value} Mpa' } }

    series: [
      {
        name: '温度 (℃)',
        type: 'line',
        symbol: 'none', // 隐藏数据点的小圆圈，极大提升渲染性能
        smooth: true,   // 开启平滑曲线
        yAxisIndex: 0,
        data: tempData,
        lineStyle: { width: 3, color: '#ff3366' }, // 炫酷红
        areaStyle: {
          color: new echarts.graphic.LinearGradient(0, 0, 0, 1, [
            { offset: 0, color: 'rgba(255, 51, 102, 0.4)' },
            { offset: 1, color: 'rgba(255, 51, 102, 0)' }
          ])
        }
      },
      {
        name: '压力 (Mpa)',
        type: 'line',
        symbol: 'none',
        smooth: true,

        data: pressData,
        lineStyle: { width: 3, color: '#00ffcc' }, // 科技绿

          ])
